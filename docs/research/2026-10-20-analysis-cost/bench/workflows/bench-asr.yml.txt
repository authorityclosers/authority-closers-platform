# Analysis-cost research (20 Oct 2026): self-hosted ASR arm (faster-whisper, CPU int8) on free runners over audio
# rendered by bench-tts runs. Runs only for pushes to bench-asr/** refs; bench/asr-request.json names the runs.
name: bench-asr

on:
  push:
    branches:
      - "bench-asr/**"

permissions:
  contents: read
  actions: read

jobs:
  plan:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    outputs:
      matrix: ${{ steps.p.outputs.matrix }}
    steps:
      - uses: actions/checkout@v4
      - id: p
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          python3 - <<'EOF' >> "$GITHUB_OUTPUT"
          import json, subprocess
          req = json.load(open("bench/asr-request.json"))
          items = []
          for run in req["runs"]:
              arts = json.loads(subprocess.check_output(["gh", "api", f"repos/${{ github.repository }}/actions/runs/{run}/artifacts", "--jq", ".artifacts"]))
              for a in arts:
                  if a["name"].startswith(("audio-", "realspeech-")):
                      for m in req["models"]:
                          items.append({"run": str(run), "artifact": a["name"], "model": m})
          print("matrix=" + json.dumps({"include": items}))
          EOF
          cat "$GITHUB_OUTPUT"

  asr:
    needs: plan
    runs-on: ubuntu-latest
    timeout-minutes: 340
    strategy:
      fail-fast: false
      max-parallel: 6   # leave the org's 20 concurrent jobs for wb-ci
      matrix: ${{ fromJson(needs.plan.outputs.matrix) }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
      - run: pip install -q faster-whisper==1.1.0 requests==2.32.3
      - uses: actions/download-artifact@v4
        with:
          name: ${{ matrix.artifact }}
          path: audio
          run-id: ${{ matrix.run }}
          github-token: ${{ github.token }}
      - uses: actions/cache@v4
        with:
          path: ~/.cache/huggingface
          key: fw-${{ matrix.model }}-v1
      - name: Transcribe
        run: python bench/asr_local.py --audio audio --model ${{ matrix.model }} --out asr
      - uses: actions/upload-artifact@v4
        with:
          name: asr-${{ matrix.model }}-${{ matrix.artifact }}
          path: asr
          retention-days: 30
